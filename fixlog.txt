# --- Google Colab Folder Tree Display Script ---

import os
import zipfile
from google.colab import files

# 1. Upload ZIP file
print("Upload your ZIP file (e.g. project.zip)...")
uploaded = files.upload()

# 2. Extract it to /home
for fname in uploaded.keys():
    zip_path = f"/home/{fname}"
    with open(zip_path, "wb") as f:
        f.write(uploaded[fname])
    extract_dir = "/home"
    with zipfile.ZipFile(zip_path, 'r') as zip_ref:
        zip_ref.extractall(extract_dir)
    print(f"Extracted to: {extract_dir}")

# 3. Find extracted folder (assuming one main root directory)
entries = [d for d in os.listdir(extract_dir) if os.path.isdir(os.path.join(extract_dir, d))]
root_dir = entries[0] if entries else extract_dir
root_path = os.path.join(extract_dir, root_dir)

print(f"\nRoot project folder detected: {root_path}\n")

# 4. Recursive function to print tree
def print_tree(startpath, prefix=""):
    items = sorted(os.listdir(startpath))
    for index, item in enumerate(items):
        path = os.path.join(startpath, item)
        connector = "└── " if index == len(items) - 1 else "├── "
        print(prefix + connector + item)
        if os.path.isdir(path):
            extension = "    " if index == len(items) - 1 else "│   "
            print_tree(path, prefix + extension)

# 5. Print tree structure
print(f"{os.path.basename(root_path)}/")
print_tree(root_path)
